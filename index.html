JAVA PROJECT

PORT SCANNER SYSTEM

A Java networking project that scans ports on a target, records open and closed ports, highlights predefined risky ports, and saves a scan report.

PortScannerSystem.java Download .java 

import java.io.FileWriter;
import java.io.PrintWriter;
import java.net.InetSocketAddress;
import java.net.Socket;
import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class PortScannerSystem {

    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);

        Map<Integer, String> riskyPorts = new HashMap<>();
        StringBuilder openPortsList = new StringBuilder();
        StringBuilder closedPortsList = new StringBuilder();
        StringBuilder vulnerablePortsList = new StringBuilder();

int vulnerableCount = 0;


        riskyPorts.put(21, "FTP - Weak authentication may allow unauthorized access.");
        riskyPorts.put(22, "SSH - Weak passwords may allow brute-force attacks.");
        riskyPorts.put(23, "Telnet - Communication is not encrypted.");
        riskyPorts.put(25, "SMTP - Can be abused for spam and email spoofing.");
        riskyPorts.put(53, "DNS - Vulnerable to amplification attacks.");
        riskyPorts.put(80, "HTTP - May expose web application vulnerabilities.");
        riskyPorts.put(8080, "Web Service - Software should be patched.");
        riskyPorts.put(8081, "Web Service -Check for misconfigurations.");
        riskyPorts.put(8082,"Web Service - Monitor for unauthorized access.");

        System.out.println("====================================");
        System.out.println("       PORT SCANNER SYSTEM");
        System.out.println("====================================");

        System.out.print("Enter Target IP Address: ");
        String target = input.nextLine();

        System.out.print("Enter Scan Speed (milliseconds): ");
        int speed = input.nextInt();

        int openCount = 0;
        int closedCount = 0;

        StringBuilder report = new StringBuilder();

        report.append("PORT SCANNER REPORT\n");
        report.append("Target: ").append(target).append("\n\n");

        System.out.println("\nStarting Scan...\n");

        for (int port = 20; port <= 150; port++) {

            System.out.print("Scanning Port " + port + " ... ");

            boolean open = false;

            try {
                Socket socket = new Socket();
                socket.connect(new InetSocketAddress(target, port), 50);
                socket.close();
                open = true;
            } catch (Exception e) {
                open = false;
            }

            if (open) {
                System.out.println("OPEN");
                openCount++;
                
                openPortsList.append("Port")
                        .append(port)
                        .append(" :OPEN\n");

                report.append("Port ")
                .append(port)
                .append(" : OPEN\n");
            } 
            else {
                System.out.println("CLOSED");
                closedCount++;

                closedPortsList.append("Port")
                     .append(port)
                     .append(" :CLOSED\n");

                report.append("Port ")
                .append(port)
                .append(" : CLOSED\n");
            }

            if (riskyPorts.containsKey(port)) {

                vulnerableCount++;

                String risk = riskyPorts.get(port);

                vulnerablePortsList.append("Port")
                              .append(port)
                              .append("\nRisk: ")
                              .append(risk)
                              .append("\n\n");

                System.out.println("   [VULNERABLE PORT]");
                System.out.println("Security Risk:   " + risk);

                report.append("VULNERABLE PORT\n ");
                report.append("port:")
                      .append(port)
                      .append("\nRisk:")
                      .append(risk)
                      .append("\n\n");
            }

            try {
                Thread.sleep(speed);
            } catch (Exception e) {
            }
        }

        System.out.println("\n====================================");
        System.out.println("SCAN COMPLETED");
        System.out.println("====================================");
        System.out.println("\n=================================");
System.out.println("OPEN PORTS");
System.out.println("=================================");
System.out.println(openPortsList);

System.out.println("\n=================================");
System.out.println("CLOSED PORTS");
System.out.println("=================================");
System.out.println(closedPortsList);

System.out.println("\n=================================");
System.out.println("VULNERABLE PORTS");
System.out.println("=================================");
System.out.println(vulnerablePortsList);

        System.out.println("Open Ports Found   : " + openCount);
        System.out.println("Closed Ports Found : " + closedCount);
        System.out.println("Vulnerable Ports Found :" + vulnerableCount);

        System.out.println("\nSecurity Summary:");
        System.out.println("- Disable unused services.");
        System.out.println("- Use strong passwords.");
        System.out.println("- Avoid Telnet where possible.");
        System.out.println("- Keep servers patched and updated.");

        report.append("\n");
        report.append("Open Ports Found: ").append(openCount).append("\n");
        report.append("Closed Ports Found: ").append(closedCount).append("\n");

        try {
            PrintWriter writer =
                new PrintWriter(new FileWriter("scan_report.txt"));

            writer.print(report.toString());
            writer.close();

            System.out.println("\nReport saved as: scan_report.txt");

        } catch (Exception e) {
            System.out.println("Failed to save report.");
        }

        input.close();
    }
}
